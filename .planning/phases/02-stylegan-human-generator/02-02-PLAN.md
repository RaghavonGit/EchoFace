        from unittest.mock import MagicMock, patch
        cpu_cfg = {'device': torch.device('cpu'), 'dtype': torch.float32,
                   'use_fp16': False, 'resolution': 256, 'truncation_psi': 0.5}
        pkl = "StyleGAN-Human/pretrained_models/stylegan_human_v2_1024.pkl"
        with patch('generator.stylegan_wrapper.legacy.load_network_pkl') as mock_load, \
             patch('generator.stylegan_wrapper.dnnlib.util.open_url') as mock_url:
            mock_G = MagicMock()
            mock_G.parameters.return_value = []
            # synthesis returns [1, 3, 1024, 1024] float32 — native model output shape
            mock_G.synthesis.return_value = torch.zeros(1, 3, 1024, 1024, dtype=torch.float32)
            mock_load.return_value = {'G_ema': mock_G}
            mock_url.return_value.__enter__ = MagicMock(return_value=MagicMock())
            mock_url.return_value.__exit__ = MagicMock(return_value=False)
            wrapper = StyleGANWrapper(pkl, cpu_cfg)
            w = torch.zeros(1, 18, 512)
            result = wrapper.synthesize(w)
