# Face Edit Feature Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a cumulative face-edit capability — after generating a face, the user can apply voice or text edit instructions (e.g. "change eyes to blue") that re-optimize from the current W+ latent, with a Reset button to restore the original.

**Architecture:** `StyleCLIPWrapper` stores W+ latent state after each generate/edit; a new `edit()` method re-optimizes from the current latent with a combined prompt and stronger L2 anchor; `reset()` restores the original latent in a single forward pass. The Gradio UI gains an Edit Details section (mic + text + Apply Edit + Reset) that appears after the first successful generation.

**Tech Stack:** PyTorch, CLIP ViT-B/32, StyleGAN2-ADA, Gradio 3.x, Python 3.8

```python
def edit(
    self,
    edit_instruction: str,
    steps: int = 60,
    lr: float = 0.05,
    callback=None,
) -> tuple:
    """
    Edit the current face by re-optimizing from the stored W+ latent.

    Args:
        edit_instruction: Natural language edit (e.g. "change eyes to blue").
        steps:            Optimization steps (default 60).
        lr:               Adam learning rate.
        callback:         Optional fn(step, loss, PIL.Image) every 10 steps.

    Returns:
        (PIL.Image, float) — updated face and CLIP similarity score.
    """
    if self._current_w is None:
        raise RuntimeError("Generate a face first before editing.")

    # Combined prompt: original description + edit instruction
    combined = (
        f"a photo of a face, {self._original_prompt}, {edit_instruction}"
    )
    print(f"[StyleCLIP] Edit prompt: '{combined}'")

    # Start from current face latent (not w_avg)
    w_opt = self._current_w.clone().requires_grad_(True)

    optimizer = torch.optim.Adam([w_opt], lr=lr)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(
        optimizer, T_max=steps
    )

    text_feat = self._encode_text(combined)   # [1, 512], no grad

    # L2 anchor: current face — stronger lambda keeps identity stable
    w_anchor = self._current_w.detach()
    l2_lambda = 0.05

    best_loss = float("inf")
    no_improve = 0

    try:
        for step in range(steps):
            optimizer.zero_grad()

            img_small = self._synthesise_small(w_opt)
            img_feat  = self._clip_encode_image(img_small.float())

            cos_loss = 1.0 - (img_feat * text_feat).sum(dim=-1).mean()
            l2_loss  = (w_opt - w_anchor).pow(2).sum().add(1e-8).sqrt()
            loss     = cos_loss + l2_lambda * l2_loss

            if torch.isnan(loss):
                print(f"  ⚠ Edit step {step}: NaN loss, skipping")
                optimizer.zero_grad()
                continue

            loss.backward()
            torch.nn.utils.clip_grad_norm_([w_opt], max_norm=1.0)
            optimizer.step()
            scheduler.step()

            if loss.item() < best_loss - 0.001:
                best_loss = loss.item()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve >= 15:
                print(f"  → Edit early stop at step {step}")
                break

            if step % 10 == 0:
                print(
                    f"  → Edit step {step:3d}/{steps} "
                    f"| CLIP loss: {cos_loss.item():.4f} "
                    f"| L2: {l2_loss.item():.2f}"
                )
                if callback is not None:
                    with torch.no_grad():
                        preview = self._synthesise_small(w_opt.detach())
                        pil_preview = Image.fromarray(
                            (preview[0]
                             .permute(1, 2, 0)
                             .clamp(0, 1)
                             .mul(255)
                             .to(torch.uint8)
                             .cpu()
                             .numpy()),
                            "RGB",
                        )
                    callback(step, loss.item(), pil_preview)

        # Final render at full resolution
        print("[StyleCLIP] Rendering edited face (1024×1024) ...")
        with torch.no_grad():
            with _quiet():
                final_raw = self.G.synthesis(
                    w_opt.detach(), noise_mode="const", force_fp32=True
                )
            final_raw = (final_raw + 1.0) * 0.5
            final_raw = torch.nn.functional.interpolate(
                final_raw, size=(1024, 1024), mode="bilinear",
                align_corners=False
            )
            final_arr = (
                final_raw[0]
                .permute(1, 2, 0)
                .clamp(0, 1)
                .mul(255)
                .to(torch.uint8)
                .cpu()
                .numpy()
            )

        final_pil = Image.fromarray(final_arr, "RGB")
        final_sim = 1.0 - cos_loss.item()

        # Update current latent for next cumulative edit
        self._current_w = w_opt.detach().clone()

        if callback is not None:
            callback(steps, final_sim, final_pil)

        return final_pil, final_sim

    except Exception as e:
        print(f"\n[CRITICAL ERROR in edit] {e}")
        traceback.print_exc()
        raise
```

```python
def reset(self) -> tuple:
    """
    Restore the face to the original generated state.

    Returns:
        (PIL.Image, float) — original face and original CLIP similarity score.
    """
    if self._original_w is None:
        raise RuntimeError("Generate a face first.")

    # Restore current latent to original
    self._current_w = self._original_w.clone()

    # Single synthesis pass — no optimization
    print("[StyleCLIP] Resetting to original face ...")
    with torch.no_grad():
        with _quiet():
            raw = self.G.synthesis(
                self._original_w, noise_mode="const", force_fp32=True
            )
        raw = (raw + 1.0) * 0.5
        raw = torch.nn.functional.interpolate(
            raw, size=(1024, 1024), mode="bilinear", align_corners=False
        )
        arr = (
            raw[0]
            .permute(1, 2, 0)
            .clamp(0, 1)
            .mul(255)
            .to(torch.uint8)
            .cpu()
            .numpy()
        )

    return Image.fromarray(arr, "RGB"), self._original_sim
```

```python
def edit_fn(mic_audio, edit_text, wrapper, cfg):
    """
    Yields 8-element tuples (same schema as generate_fn):
      0  status_out   (str)
      1  edit_text_in (str)
      2  score_out    (str)
      3  image_out    (np.ndarray | None)
      4  stored_image (PIL.Image | None)
      5  apply_btn    (gr.update)
      6  loader_html  (gr.update)
      7  edit_section (gr.update — no change)
    """
    def _y(status, text, score, img, pil, btn, loader):
        return (status, text, score, img, pil, btn,
                gr.update(value=loader), gr.update())

    prompt = edit_text.strip() if edit_text else ""

    # ── Guard: nothing provided ────────────────────────────────────────────
    if not prompt and mic_audio is None:
        yield _y("Please provide an edit instruction or record audio.",
                 "", "", None, None, gr.update(visible=True), _LOADER_OFF)
        return

    # ── Stage 1: ASR ──────────────────────────────────────────────────────
    if not prompt and mic_audio is not None:
        yield _y("Transcribing...", "", "", None, None,
                 gr.update(visible=False), _LOADER_ON)
        audio_array   = process_audio(mic_audio)
        transcription = Transcriber(cfg).transcribe(audio_array)
        yield _y("Transcription complete. Review and click Apply Edit again.",
                 transcription, "", None, None,
                 gr.update(visible=True), _LOADER_OFF)
        return

    # ── Guard: prompt too short ────────────────────────────────────────────
    if len(prompt.split()) < 4:
        yield _y("⚠ Edit too short — be specific (e.g. 'change eyes to blue').",
                 prompt, "", None, None, gr.update(visible=True), _LOADER_OFF)
        return

    # ── Stage 2: Edit optimization ─────────────────────────────────────────
    score_q       = queue.Queue()
    result_holder = {}
    t_start       = time.time()

    def _cb(step, loss, pil_img):
        score_q.put(("progress", step, loss, pil_img))

    def _run():
        try:
            pil_image, final_sim = wrapper.edit(prompt, callback=_cb)
            result_holder["success"] = (pil_image, final_sim)
            score_q.put(("done", None, None, None))
        except Exception as exc:
            import traceback; traceback.print_exc()
            result_holder["error"] = exc
            score_q.put(("error", None, None, None))

    threading.Thread(target=_run, daemon=True).start()

    last_img_arr = None

    while True:
        msg, step, loss, pil_img = score_q.get()

        if msg == "error":
            yield _y(f"Edit failed: {result_holder['error']}",
                     "", "", None, None, gr.update(visible=True), _LOADER_OFF)
            return

        elif msg == "done":
            elapsed   = time.time() - t_start
            pil_image, final_sim = result_holder["success"]
            final_arr = np.array(pil_image) if pil_image is not None else last_img_arr
            yield _y(f"Edit complete.  {elapsed:.0f}s",
                     "", f"CLIP Score: {final_sim:.4f}",
                     final_arr, pil_image, gr.update(visible=True), _LOADER_OFF)
            return

        elif msg == "progress":
            elapsed = time.time() - t_start
            if pil_img is not None:
                last_img_arr = np.array(pil_img)
            yield _y(f"Applying edit...  step {step} / 60  ({elapsed:.0f}s)",
                     prompt, f"Loss: {loss:.4f}",
                     last_img_arr, pil_img,
                     gr.update(visible=False), _LOADER_ON)
```

```python
def reset_fn(wrapper):
    """
    Restore the face to its originally generated state.

    Returns 5-tuple: (image_arr, status, score, stored_pil, export_btn)
    """
    try:
        pil_image, sim = wrapper.reset()
        return (
            np.array(pil_image),
            "Reset to original.",
            f"CLIP Score: {sim:.4f}",
            pil_image,
```
